cmake_minimum_required(VERSION 3.16)
project(pr2_latency_template VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(project_options INTERFACE)
if(MSVC)
    target_compile_options(project_options INTERFACE /W4 /utf-8)
else()
    target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic)
    if(ENABLE_SANITIZERS)
        target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(project_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

foreach(module protocol telemetry transport)
    add_library(${module} src/${module}.cpp)
    target_include_directories(${module} PUBLIC include)
    target_link_libraries(${module} PUBLIC project_options)
endforeach()
if(WIN32)
    target_link_libraries(transport PRIVATE ws2_32)
endif()
add_library(csv_log src/csv_log.cpp)
target_link_libraries(csv_log PUBLIC telemetry)
add_executable(server src/server_main.cpp)
target_link_libraries(server PRIVATE protocol transport)
add_executable(client src/client_main.cpp)
target_link_libraries(client PRIVATE protocol telemetry transport csv_log)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
